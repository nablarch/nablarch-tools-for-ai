# DBからファイル出力パターン（生成AI用）

```java
public class DbToFileAction extends BatchAction<SqlRow> {
    private ObjectMapper<OutputDto> mapper;

    @Override
    protected void initialize(CommandLine command, ExecutionContext context) {
        FilePathSetting setting = FilePathSetting.getInstance();
        File file = setting.getFile("output", "fileName");
        mapper = ObjectMapperFactory.create(OutputDto.class, new FileOutputStream(file));
    }

    @Override
    public DataReader<SqlRow> createReader(ExecutionContext context) {
        DatabaseRecordReader reader = new DatabaseRecordReader();
        SqlPStatement statement = getSqlPStatement("FIND_DATA");
        String bizDate = BusinessDateUtil.getDate();
        statement.setString(1, bizDate);
        reader.setStatement(statement);
        return reader;
    }

    @Override
    public Result handle(SqlRow record, ExecutionContext context) {
        OutputDto dto = EntityUtil.createEntity(OutputDto.class, record);
        mapper.write(dto);
        return new Success();
    }

    @Override
    protected void terminate(Result result, ExecutionContext context) {
        mapper.close();
    }
}

// 出力DTO
@Csv(properties = {"id", "name", "date"}, type = CsvType.CUSTOM)
@CsvFormat(charset = "UTF-8", fieldSeparator = ',', lineSeparator = "\r\n")
public class OutputDto {
    private String id;
    private String name; 
    private String date;
    // getter/setter省略
}
```
