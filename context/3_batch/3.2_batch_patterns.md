// ファイル読み込み
public class FileDataReader implements DataReader<FormClass> {
    private ObjectMapperIterator<FormClass> iterator;
    
    public FormClass read(ExecutionContext ctx) {
        if (iterator == null) initialize();
        return iterator.next();
    }
    
    public boolean hasNext(ExecutionContext ctx) {
        if (iterator == null) initialize();
        return iterator.hasNext();
    }
    
    public void close(ExecutionContext ctx) {
        if (iterator != null) iterator.close();
    }
    
    private void initialize() {
        FilePathSetting setting = FilePathSetting.getInstance();
        File file = setting.getFileWithoutCreate("input", "fileName");
        iterator = new ObjectMapperIterator<>(
            ObjectMapperFactory.create(FormClass.class, new FileInputStream(file))
        );
    }
}

## 3. DB to DB パターン（データベース → データベース）

### 基本実装

```java
public class DbToDbAction extends BatchAction<SqlRow> {

    @Override
    public DataReader<SqlRow> createReader(ExecutionContext context) {
        DatabaseRecordReader reader = new DatabaseRecordReader();
        SqlPStatement statement = getSqlPStatement("SELECT_SOURCE_DATA");
        reader.setStatement(statement);
        return reader;
    }

    @Override
    public Result handle(SqlRow inputData, ExecutionContext context) {
        // データ変換処理
        TargetEntity entity = convertToTargetEntity(inputData);
        
        // データベース操作
        if (existsTarget(entity.getId())) {
            UniversalDao.update(entity);
        } else {
            UniversalDao.insert(entity);
        }
        
        return new Success();
    }
    
    private TargetEntity convertToTargetEntity(SqlRow sourceData) {
        TargetEntity entity = EntityUtil.createEntity(TargetEntity.class, sourceData);
        // 追加変換処理
        entity.setProcessedDate(BusinessDateUtil.getDate());
        return entity;
    }
    
    private boolean existsTarget(String id) {
        return UniversalDao.exists(TargetEntity.class, "id", id);
    }
}
```

## 4. バッチ更新パターン（大量データ処理）

### 効率的な大量更新

```java
public class BatchUpdateAction extends BatchAction<SqlRow> {
    private List<UpdateEntity> batchList = new ArrayList<>();
    private static final int BATCH_SIZE = 1000;

    @Override
    public DataReader<SqlRow> createReader(ExecutionContext context) {
        DatabaseRecordReader reader = new DatabaseRecordReader();
        SqlPStatement statement = getSqlPStatement("SELECT_UPDATE_TARGET");
        reader.setStatement(statement);
        return reader;
    }

    @Override
    public Result handle(SqlRow inputData, ExecutionContext context) {
        // エンティティ作成
        UpdateEntity entity = EntityUtil.createEntity(UpdateEntity.class, inputData);
        entity.setUpdateDate(BusinessDateUtil.getDate());
        
        batchList.add(entity);
        
        // バッチサイズに達したら一括更新
        if (batchList.size() >= BATCH_SIZE) {
            UniversalDao.batchUpdate(batchList);
            batchList.clear();
        }
        
        return new Success();
    }
    
    @Override
    protected void terminate(Result result, ExecutionContext context) {
        // 残りのデータを一括更新
        if (!batchList.isEmpty()) {
            UniversalDao.batchUpdate(batchList);
            batchList.clear();
        }
    }
}
```

## 5. ファイル分割出力パターン

### 条件別ファイル分割

```java
public class FileSplitAction extends BatchAction<SqlRow> {
    private Map<String, ObjectMapper<OutputDto>> mappers = new HashMap<>();
    private FilePathSetting filePathSetting;

    @Override
    protected void initialize(CommandLine command, ExecutionContext context) {
        filePathSetting = FilePathSetting.getInstance();
    }

    @Override
    public DataReader<SqlRow> createReader(ExecutionContext context) {
        DatabaseRecordReader reader = new DatabaseRecordReader();
        SqlPStatement statement = getSqlPStatement("SELECT_ALL_DATA");
        reader.setStatement(statement);
        return reader;
    }

    @Override
    public Result handle(SqlRow inputData, ExecutionContext context) {
        String category = inputData.getString("CATEGORY");
        
        // カテゴリ別にマッパー取得（初回作成）
        ObjectMapper<OutputDto> mapper = getOrCreateMapper(category);
        
        OutputDto dto = EntityUtil.createEntity(OutputDto.class, inputData);
        mapper.write(dto);
        
        return new Success();
    }
    
    private ObjectMapper<OutputDto> getOrCreateMapper(String category) {
        return mappers.computeIfAbsent(category, cat -> {
            try {
                File outputFile = filePathSetting.getFile("csv-output", "output_" + cat);
                FileOutputStream fos = new FileOutputStream(outputFile);
                return ObjectMapperFactory.create(OutputDto.class, fos);
            } catch (FileNotFoundException e) {
                throw new IllegalStateException(e);
            }
        });
    }
    
    @Override
    protected void terminate(Result result, ExecutionContext context) {
        // 全マッパーをクローズ
        mappers.values().forEach(ObjectMapper::close);
    }
}
```

### エラーレコード別ファイル出力

public class ErrorHandlingAction extends BatchAction<SqlRow> {
    private ObjectMapper<OutputDto> successMapper;
    private ObjectMapper<ErrorDto> errorMapper;
    private AtomicInteger successCount = new AtomicInteger(0);
    private AtomicInteger errorCount = new AtomicInteger(0);

    @Override
    protected void initialize(CommandLine command, ExecutionContext context) {
        FilePathSetting filePathSetting = FilePathSetting.getInstance();
        try {
            File successFile = filePathSetting.getFile("csv-output", "success");
            File errorFile = filePathSetting.getFile("csv-output", "error");
            
            successMapper = ObjectMapperFactory.create(
                OutputDto.class, new FileOutputStream(successFile));
            errorMapper = ObjectMapperFactory.create(
                ErrorDto.class, new FileOutputStream(errorFile));
        } catch (FileNotFoundException e) {
            throw new IllegalStateException(e);
        }
    }

    @Override
    public DataReader<SqlRow> createReader(ExecutionContext context) {
        DatabaseRecordReader reader = new DatabaseRecordReader();
        SqlPStatement statement = getSqlPStatement("SELECT_PROCESS_DATA");
        reader.setStatement(statement);
        return reader;
    }

    @Override
    public Result handle(SqlRow inputData, ExecutionContext context) {
        try {
            // 業務処理
            processBusinessLogic(inputData);
            
            // 成功データを出力
            OutputDto dto = EntityUtil.createEntity(OutputDto.class, inputData);
            dto.setProcessStatus("SUCCESS");
            successMapper.write(dto);
            successCount.incrementAndGet();
            
            return new Success();
            
        } catch (Exception e) {
            // エラーデータを出力
            ErrorDto errorDto = new ErrorDto();
            errorDto.setId(inputData.getString("ID"));
            errorDto.setErrorMessage(e.getMessage());
            errorDto.setProcessDate(BusinessDateUtil.getDate().toString());
            errorMapper.write(errorDto);
            errorCount.incrementAndGet();
            
            // エラーレコードは継続処理
            return new Success();
        }
    }
    
    private void processBusinessLogic(SqlRow inputData) throws Exception {
        // 業務処理実装
        // 例外が発生する可能性のある処理
    }
    
    @Override
    protected void terminate(Result result, ExecutionContext context) {
        if (successMapper != null) successMapper.close();
        if (errorMapper != null) errorMapper.close();
        
        // 処理件数ログ出力
        LOGGER.info("処理完了 - 成功件数: {}, エラー件数: {}", 
                   successCount.get(), errorCount.get());
    }
}

        } catch (NoDataException e) {
            // データが存在しない（他プロセスで処理済み）
            return new Success();
        }

これらのパターンを参考に、Nablarchフレームワークの設計思想に沿ったバッチアプリケーションを実装してください。特に重要なのは：

1. **フレームワーク制御**: 自前でループを書かない
2. **適切なリソース管理**: initialize/terminateの活用
3. **データリーダーの責務分離**: 読み込み専用クラスとして実装
4. **標準機能の活用**: UniversalDao、EntityUtil、BusinessDateUtilなどを使用
