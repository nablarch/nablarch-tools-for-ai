# ファイル変換パターン（生成AI用）

```java
// ファイル変換アクション
public class FileToFileAction extends BatchAction<InputForm> {
    private ObjectMapper<OutputDto> outputMapper;

    @Override
    protected void initialize(CommandLine command, ExecutionContext context) {
        FilePathSetting setting = FilePathSetting.getInstance();
        File file = setting.getFile("output", "fileName");
        outputMapper = ObjectMapperFactory.create(OutputDto.class, new FileOutputStream(file));
    }

    @Override
    public DataReader<InputForm> createReader(ExecutionContext context) {
        return new InputFileReader();
    }

    @Override
    public Result handle(InputForm inputData, ExecutionContext context) {
        // データ変換
        OutputDto output = BeanUtil.createAndCopy(OutputDto.class, inputData);
        output.setProcessedDate(BusinessDateUtil.getDate());
        output.setStatus("PROCESSED");
        
        outputMapper.write(output);
        return new Success();
    }
    
    @Override
    protected void terminate(Result result, ExecutionContext context) {
        outputMapper.close();
    }
}

// 入力ファイルリーダー
public class InputFileReader implements DataReader<InputForm> {
    private ObjectMapperIterator<InputForm> iterator;

    @Override
    public InputForm read(ExecutionContext ctx) {
        if (iterator == null) initialize();
        return iterator.next();
    }

    @Override
    public boolean hasNext(ExecutionContext ctx) {
        if (iterator == null) initialize();
        return iterator.hasNext();
    }

    @Override
    public void close(ExecutionContext ctx) {
        if (iterator != null) iterator.close();
    }

    private void initialize() {
        FilePathSetting setting = FilePathSetting.getInstance();
        File file = setting.getFileWithoutCreate("input", "fileName");
        iterator = new ObjectMapperIterator<>(
            ObjectMapperFactory.create(InputForm.class, new FileInputStream(file)));
    }
}

public class OutputDto {
    private String id;

    private String amount;

    private String status;
    // getter/setter省略

## 重要ポイント

- 入力：`ObjectMapperIterator`でファイル読み込み
- 出力：`initialize()`/`terminate()`でファイル管理
- `BeanUtil.createAndCopy()`でデータ変換
- 両方のDTO定義に`@Csv`アノテーション必須
