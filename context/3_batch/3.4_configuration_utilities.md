## ObjectMapper使用例

```java
// ファイル出力用
public class FileOutputAction extends BatchAction<SqlRow> {
    private ObjectMapper<OutputDto> mapper;
    
    @Override
    protected void initialize(CommandLine command, ExecutionContext context) {
        FilePathSetting setting = FilePathSetting.getInstance();
        File file = setting.getFile("output", "fileName");
        mapper = ObjectMapperFactory.create(OutputDto.class, new FileOutputStream(file));
    }
    
    @Override
    public Result handle(SqlRow inputData, ExecutionContext context) {
        OutputDto dto = EntityUtil.createEntity(OutputDto.class, inputData);
        mapper.write(dto);
        return new Success();
    }
    
    @Override
    protected void terminate(Result result, ExecutionContext context) {
        mapper.close();
    }
}

// ファイル読み込み用
public class FileDataReader implements DataReader<FormClass> {
    private ObjectMapperIterator<FormClass> iterator;
    
    @Override
    public FormClass read(ExecutionContext ctx) {
        if (iterator == null) initialize();
        return iterator.next();
    }
    
    @Override
    public boolean hasNext(ExecutionContext ctx) {
        if (iterator == null) initialize();
        return iterator.hasNext();
    }
    
    @Override
    public void close(ExecutionContext ctx) {
        if (iterator != null) iterator.close();
    }
    
    private void initialize() {
        FilePathSetting setting = FilePathSetting.getInstance();
        File file = setting.getFileWithoutCreate("input", "fileName");
        iterator = new ObjectMapperIterator<>(
            ObjectMapperFactory.create(FormClass.class, new FileInputStream(file))
        );
    }
}
```

```java
    @NumberRange(min = 0, max = 10000000)
    private String salary;
    
    /** 部署ID */
    @Required
    @Length(max = 10)
    private String departmentId;
    
    /** 入社日 */
    @Required
    @DateString(format = "yyyy/MM/dd")
    private String hireDate;
    
    /** 行番号（自動設定） */
    @LineNumber
    private Long lineNumber;
    
    // getter/setter省略
    
    /**
     * 相関バリデーション例
     */
    @ValidateFor("register")
    public static void validateEmployeeData(ValidationContext<EmployeeImportForm> context) {
        EmployeeImportForm form = context.createObject();
        
        // 部署ID存在チェック
        if (StringUtil.hasValue(form.getDepartmentId())) {
            boolean departmentExists = UniversalDao.exists(
                DepartmentEntity.class, "departmentId", form.getDepartmentId());
            if (!departmentExists) {
                context.addResultMessage("departmentId", "部署IDが存在しません", form.getDepartmentId());
            }
        }
        
        // 従業員ID重複チェック
        if (StringUtil.hasValue(form.getEmployeeId())) {
            boolean employeeExists = UniversalDao.exists(
                EmployeeEntity.class, "employeeId", form.getEmployeeId());
            if (employeeExists) {
                context.addResultMessage("employeeId", "従業員IDが既に存在します", form.getEmployeeId());
            }
        }
    }
}
```

```java
public class LoggingExample extends BatchAction<SqlRow> {
    private static final Logger LOGGER = LoggerManager.get(LoggingExample.class);
    
    private AtomicInteger processedCount = new AtomicInteger(0);
    private AtomicInteger errorCount = new AtomicInteger(0);
    
    @Override
    protected void initialize(CommandLine command, ExecutionContext context) {
        LOGGER.logInfo("バッチ処理開始 - リクエストID: {}", context.getRequestId());
        // 初期化処理
    }
    
    @Override
    public Result handle(SqlRow record, ExecutionContext context) {
        try {
            // 業務処理
            String id = record.getString("ID");
            LOGGER.logDebug("処理中 - ID: {}", id);
            
            processRecord(record);
            processedCount.incrementAndGet();
            
            // 100件ごとに進捗ログ出力
            int count = processedCount.get();
            if (count % 100 == 0) {
                LOGGER.logInfo("処理進捗 - 処理件数: {}", count);
            }
            
            return new Success();
            
        } catch (Exception e) {
            errorCount.incrementAndGet();
            String id = record.getString("ID");
            LOGGER.logError("処理エラー - ID: {}, エラー: {}", id, e.getMessage(), e);
            
            // エラー時も処理継続
            return new Success();
        }
    }
    
    @Override
    protected void terminate(Result result, ExecutionContext context) {
        LOGGER.logInfo("バッチ処理完了 - 処理件数: {}, エラー件数: {}", 
                      processedCount.get(), errorCount.get());
        
        // リソース解放等
    }
    
    private void processRecord(SqlRow record) {
        // 業務処理実装
    }
}
```
