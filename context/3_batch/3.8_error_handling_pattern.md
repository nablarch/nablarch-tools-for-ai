**前提**: 3.1_form_dto_basics.md（DTO定義）を読み込み済み

```java
public class ErrorHandlingAction extends BatchAction<SqlRow> {
    private ObjectMapper<OutputDto> successMapper;
    private ObjectMapper<ErrorDto> errorMapper;
    private AtomicInteger successCount = new AtomicInteger(0);
    private AtomicInteger errorCount = new AtomicInteger(0);

    @Override
    protected void initialize(CommandLine command, ExecutionContext context) {
        FilePathSetting setting = FilePathSetting.getInstance();
        File successFile = setting.getFile("output", "success");
        File errorFile = setting.getFile("output", "error");
        
        successMapper = ObjectMapperFactory.create(OutputDto.class, new FileOutputStream(successFile));
        errorMapper = ObjectMapperFactory.create(ErrorDto.class, new FileOutputStream(errorFile));
    }

    @Override
    public DataReader<SqlRow> createReader(ExecutionContext context) {
        DatabaseRecordReader reader = new DatabaseRecordReader();
        SqlPStatement statement = getSqlPStatement("FIND_PROCESS_DATA");
        reader.setStatement(statement);
        return reader;
    }

    @Override
    public Result handle(SqlRow inputData, ExecutionContext context) {
        try {
            // 業務処理
            processRecord(inputData);
            
            // 成功データ出力
            OutputDto dto = EntityUtil.createEntity(OutputDto.class, inputData);
            dto.setStatus("SUCCESS");
            successMapper.write(dto);
            successCount.incrementAndGet();
            
        } catch (Exception e) {
            // エラーデータ出力
            ErrorDto errorDto = new ErrorDto();
            errorDto.setId(inputData.getString("ID"));
            errorDto.setErrorMessage(e.getMessage());
            errorDto.setProcessDate(BusinessDateUtil.getDate());
            errorMapper.write(errorDto);
            errorCount.incrementAndGet();
            
            LOGGER.warn("処理エラー ID:{}, エラー:{}", 
                       inputData.getString("ID"), e.getMessage());
        }
        
        return new Success(); // 常に処理継続
    }
    
    @Override
    protected void terminate(Result result, ExecutionContext context) {
        if (successMapper != null) successMapper.close();
        if (errorMapper != null) errorMapper.close();
        
        LOGGER.info("処理完了 成功:{}, エラー:{}", successCount.get(), errorCount.get());
    }
    
    private void processRecord(SqlRow inputData) {
        // 業務ロジック（例外発生可能性あり）
        String value = inputData.getString("VALUE");
        if (value == null || value.isEmpty()) {
            throw new IllegalArgumentException("VALUE is required");
        }
        
        // DB更新など
        TargetEntity entity = EntityUtil.createEntity(TargetEntity.class, inputData);
        UniversalDao.insert(entity);
    }
}

    private String id;

    // getter/setter省略
}
```
