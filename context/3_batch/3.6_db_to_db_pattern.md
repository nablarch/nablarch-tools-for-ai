# DBからDB処理（専用）

**前提**: 3.1_form_dto_basics.md（Entity定義）を読み込み済み

```java
// 基本パターン
public class DbToDbAction extends BatchAction<SqlRow> {
    @Override
    public DataReader<SqlRow> createReader(ExecutionContext context) {
        DatabaseRecordReader reader = new DatabaseRecordReader();
        SqlPStatement statement = getSqlPStatement("FIND_SOURCE_DATA");
        reader.setStatement(statement);
        return reader;
    }

    @Override
    public Result handle(SqlRow inputData, ExecutionContext context) {
        TargetEntity entity = EntityUtil.createEntity(TargetEntity.class, inputData);
        entity.setProcessedDate(BusinessDateUtil.getDate());
        
        if (UniversalDao.exists(TargetEntity.class, "id", entity.getId())) {
            UniversalDao.update(entity);
        } else {
            UniversalDao.insert(entity);
        }
        return new Success();
    }
}

// 大量データパターン
public class BatchProcessAction extends BatchAction<SqlRow> {
    private List<TargetEntity> batchList = new ArrayList<>();
    private static final int BATCH_SIZE = 1000;

    @Override
    public DataReader<SqlRow> createReader(ExecutionContext context) {
        DatabaseRecordReader reader = new DatabaseRecordReader();
        SqlPStatement statement = getSqlPStatement("FIND_UPDATE_DATA");
        reader.setStatement(statement);
        return reader;
    }

    @Override
    public Result handle(SqlRow inputData, ExecutionContext context) {
        TargetEntity entity = EntityUtil.createEntity(TargetEntity.class, inputData);
        batchList.add(entity);
        
        if (batchList.size() >= BATCH_SIZE) {
            UniversalDao.batchUpdate(batchList);
            batchList.clear();
        }
        return new Success();
    }
    
    @Override
    protected void terminate(Result result, ExecutionContext context) {
        if (!batchList.isEmpty()) {
            UniversalDao.batchUpdate(batchList);
        }
    }
}
```
