# ファイルからDB登録（専用）

**前提**: 3.1_form_dto_basics.md（Form・Entity定義）と 3.2_reader_patterns.md（カスタムDataReader）を読み込み済み

```java
public class FileToDbAction extends BatchAction<ImportForm> {
    @Override
    public DataReader<ImportForm> createReader(ExecutionContext context) {
        return new CustomFileDataReader(); // データバインディング使用
    }

    @Override
    @ValidateData
    public Result handle(ImportForm inputData, ExecutionContext context) {
        DataEntity entity = BeanUtil.createAndCopy(DataEntity.class, inputData);
        UniversalDao.insert(entity);
        return new Success();
    }
}

// カスタムDataReader（データバインディング使用）
public class CustomFileDataReader implements DataReader<ImportForm> {
    private ObjectMapperIterator<ImportForm> iterator;

    @Override
    public ImportForm read(ExecutionContext ctx) {
        if (iterator == null) initialize();
        return iterator.next();
    }

    @Override
    public boolean hasNext(ExecutionContext ctx) {
        if (iterator == null) initialize();
        return iterator.hasNext();
    }

    @Override
    public void close(ExecutionContext ctx) {
        if (iterator != null) iterator.close();
    }

    private void initialize() {
        FilePathSetting setting = FilePathSetting.getInstance();
        File file = setting.getFileWithoutCreate("input", "fileName");
        try {
            iterator = new ObjectMapperIterator<>(
                ObjectMapperFactory.create(ImportForm.class, new FileInputStream(file)));
        } catch (FileNotFoundException e) {
            throw new IllegalStateException(e);
        }
    }
}
```
