# 4.3 JSP（画面表示）

## タグライブラリ宣言

```jsp
<%@ taglib prefix="c" uri="jakarta.tags.core" %>
<%@ taglib prefix="n" uri="http://tis.co.jp/nablarch" %>
```

## 重要ルール

- デフォルトでHTMLエスケープ（XSS対策）
- `nablarch_`プレフィックスは使用禁止
- name属性: リクエストスコープ変数名.プロパティ名
- 検索順序: Page→Request→RequestParam→Session

## フォーム

```jsp
<!-- 基本 -->
<n:form>内容</n:form>

<!-- 属性: method(POST/GET) action name useToken(CSRF) enctype(multipart/form-data) -->
<n:form method="POST" useToken="true">...</n:form>
```

## 入力項目

### テキスト系
```jsp
<!-- n:text 必須:name 主要:cssClass errorCss maxlength placeholder disabled readonly -->
<n:text name="form.clientName" cssClass="form-control" errorCss="input-error" maxlength="64" />

<!-- n:password 同上 -->
<n:password name="form.password" cssClass="form-control" />

<!-- n:textarea 追加:rows cols -->
<n:textarea name="form.note" rows="5" cssClass="form-control" />
```

### 選択項目
```jsp
<!-- n:select 必須:name 主要:listName elementValueProperty elementLabelProperty withNoneOption cssClass errorCss -->
<n:select name="form.industryCode" listName="industries" 
    elementValueProperty="code" elementLabelProperty="name" 
    withNoneOption="true" cssClass="form-select" />

<!-- n:radioButtons 必須:name listName elementValueProperty elementLabelProperty -->
<n:radioButtons name="form.type" listName="types" 
    elementValueProperty="value" elementLabelProperty="label" />

<!-- n:radio 必須:name value 主要:label cssClass -->
<n:radio name="form.type" value="1" label="開発" />

<!-- n:checkboxes 同radioButtons -->
<n:checkboxes name="form.classes" listName="classes" 
    elementValueProperty="value" elementLabelProperty="label" />

<!-- n:checkbox 必須:name 主要:value offValue label cssClass -->
<n:checkbox name="form.agreement" value="1" offValue="0" label="同意" />

<!-- n:hidden 必須:name -->
<n:hidden name="form.projectId" />

<!-- n:file 必須:name 主要:cssClass accept multiple -->
<n:file name="form.file" accept=".pdf" multiple="true" />
```

## ボタン・リンク

```jsp
<!-- n:button 必須:uri 主要:name cssClass disabled allowDoubleSubmission -->
<n:button uri="/action/confirm" cssClass="btn btn-success">登録</n:button>

<!-- n:submit 主要:uri type value cssClass -->
<n:submit uri="/action/create" value="登録" />

<!-- n:a 必須:href 主要:cssClass target -->
<n:a href="/action/list">一覧</n:a>

<!-- パラメータ追加: n:param 必須:paramName, name/value -->
<n:button uri="/action/show">
    <n:param paramName="id" name="row.id" />
    詳細
</n:button>
```

## 値の表示

```jsp
<!-- n:write 必須:name 主要:withHtmlFormat -->
<n:write name="project.name" />
<n:write name="project.note" withHtmlFormat="true" />

<!-- 関数 -->
${n:br(text)}
${n:formatDate(date, 'yyyy/MM/dd')}
${n:format('currency', amount)}

<!-- n:code 必須:codeId name -->
<n:code codeId="PROJECT_TYPE" name="project.type" />

<!-- n:codeSelect/n:codeRadioButtons/n:codeCheckboxes 必須:codeId name -->
<n:codeSelect codeId="PROJECT_TYPE" name="form.type" withNoneOption="true" />
```

## エラー表示

```jsp
<!-- n:error 必須:name -->
<n:error errorCss="message-error" name="form.clientName" />

<!-- n:errors 全エラー一覧 -->
<n:errors errorCss="alert alert-danger" />
```

## 確認画面パターン

```jsp
<!-- 入力画面(create.jsp) -->
<n:form>
    <n:text name="form.name" />
    <n:forInputPage>
        <n:button uri="/action/confirm">確認</n:button>
    </n:forInputPage>
    <n:forConfirmationPage>
        <n:button uri="/action/back">戻る</n:button>
        <n:button uri="/action/create" allowDoubleSubmission="false">確定</n:button>
    </n:forConfirmationPage>
</n:form>

<!-- 確認画面(confirm.jsp) -->
<n:confirmationPage path="./create.jsp" />

<!-- 非表示項目 -->
<n:ignoreConfirmation>...</n:ignoreConfirmation>
```

## 一覧・ページング

```jsp
<c:forEach var="row" items="${list}" varStatus="status">
    <tr>
        <td>${status.count}</td>
        <td><n:write name="row.name" /></td>
    </tr>
</c:forEach>

<!-- n:pagination 必須:listSearchInfoName searchUri -->
<n:pagination listSearchInfoName="searchInfo" searchUri="/action/list" />
```

## JSTL条件・繰り返し

```jsp
<!-- 条件 -->
<c:if test="${not empty list}">...</c:if>
<c:choose>
    <c:when test="${type == '1'}">開発</c:when>
    <c:otherwise>その他</c:otherwise>
</c:choose>

<!-- 繰り返し varStatus: index count first last -->
<c:forEach var="item" items="${list}" varStatus="status">
    ${status.count}: <n:write name="item.name" />
</c:forEach>
```

## セキュリティ

```jsp
<!-- CSRF自動付与 -->
<n:form useToken="true">...</n:form>

<!-- キャッシュ防止 -->
<n:noCache />

<!-- CSP対応 -->
<script nonce="<n:cspNonce />">...</script>
```

## 二重サブミット防止

```jsp
<!-- クライアント側 -->
<n:button uri="/action/create" allowDoubleSubmission="false">確定</n:button>

<!-- サーバ側: useToken="true" + Actionに@OnDoubleSubmission -->
```

## その他

```jsp
<!-- n:set 必須:var, name/value 主要:scope -->
<n:set var="title" value="顧客登録" />

<!-- n:message 必須:messageId -->
<n:message messageId="MSG00001" option0="${name}" />

<!-- n:include 必須:path -->
<n:include path="/WEB-INF/view/common/header.jsp" />

<!-- n:img 必須:src -->
<n:img src="/images/logo.png" alt="ロゴ" />

<!-- n:script 必須:src -->
<n:script src="/js/app.js" />

<!-- n:link 必須:href -->
<n:link href="/css/style.css" />
```

## name属性アクセスルール

```jsp
<!-- ドット記法 -->
<n:write name="project.client.name" />

<!-- 角括弧記法 -->
<n:write name="projects[0].name" />
<n:write name="params[key]" />
```

## 動的属性（HTML5）

```jsp
<!-- data属性・aria属性など自由に指定可能 -->
<n:button uri="/action/delete" 
    data-id="${id}" 
    data-confirm="削除しますか？"
    aria-label="削除">削除</n:button>

<!-- 論理属性: disabled readonly required multiple checked等 -->
<n:text name="form.id" disabled="true" />
```

## @InjectFormとの連携

```java
@InjectForm(form = ClientForm.class, prefix = "form")
```
```jsp
<n:text name="form.clientName" />
<n:text name="form.industryCode" />
```

## GETメソッド使用時の制約

ボタン/リンクのカスタムタグ使用不可（通常のHTMLタグ使用）

```jsp
<n:form method="GET" action="search">
    <input type="submit" value="検索" />
</n:form>
```
