# 3.7 ファイル出力パターン

## BatchAction内でのファイル出力

```java
public class FileOutputAction extends BatchAction<InputType> {
    private ObjectMapper<OutputDto> mapper;

    protected void initialize(CommandLine command, ExecutionContext context) {
        FilePathSetting filePathSetting = FilePathSetting.getInstance();
        File file = filePathSetting.getFile("csv-output", "fileName");
        try {
            mapper = ObjectMapperFactory.create(OutputDto.class, new FileOutputStream(file));
        } catch (FileNotFoundException e) {
            throw new IllegalStateException("Failed to create output file", e);
        }
    }

    public Result handle(SqlRow inputData, ExecutionContext context) {
        // 変換: SqlRow→DTO: EntityUtil.createEntity()、Form→DTO: BeanUtil.createAndCopy()
        OutputDto dto = EntityUtil.createEntity(OutputDto.class, inputData);
        dto.setProcessedDate(BusinessDateUtil.getDate());
        
        mapper.write(dto);
        return new Success();
    }

    protected void terminate(Result result, ExecutionContext context) {
        mapper.close();
    }
}
```
